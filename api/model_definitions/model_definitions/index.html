    class CNN7_Shi(nn.Module):
    """Construct the CNN7 image classifier proposed by Shi et al. (2021).

    The architecture is commonly used to compare IBP-based certified training
    methods. Dataset normalisation belongs in the loader. downsample_final=True
    provides the variant with a downsampled final convolutional stage.

    Args:
        in_shape (tuple): Shape of the input images.
        width (int): Number of channels for the convolutional layers.
        linear_size (int): Size of the fully connected layer.
        n_classes (int): Number of output classes.
        downsample_final (bool): Use stride 2 in the final convolution (TinyImageNet CNN7).

    Attributes:
        layers (nn.Sequential): Sequential container of the layers in the network.

    Methods:
        forward(x):
            Defines the forward pass of the network.
    """
    def __init__(self, in_shape=(1, 28, 28), width=64, linear_size=512, n_classes=10, downsample_final=False):
        super(CNN7_Shi, self).__init__()
        assert in_shape[1] == in_shape[2], "We only support square inputs for now!"
        in_channels = in_shape[0]
        in_dim  = in_shape[1]
        final_stride = 2 if downsample_final else 1
        feature_dim = (in_dim + 2 * final_stride - 1) // (2 * final_stride)

        self.layers = nn.Sequential(
            nn.Conv2d(in_channels, width, kernel_size=(3,3), stride=1, padding=1),
            nn.BatchNorm2d(width),
            nn.ReLU(),
            nn.Conv2d(width, width, kernel_size=(3,3), stride=1, padding=1),
            nn.BatchNorm2d(width),
            nn.ReLU(),
            nn.Conv2d(width, 2 * width, kernel_size=(3,3), stride=2, padding=1),
            nn.BatchNorm2d(2 * width),
            nn.ReLU(),
            nn.Conv2d(2 * width, 2 * width, kernel_size=(3,3), stride=1, padding=1),
            nn.BatchNorm2d(2 * width),
            nn.ReLU(),
            nn.Conv2d(2 * width, 2 * width, kernel_size=(3,3), stride=final_stride, padding=1),
            nn.BatchNorm2d(2 * width),
            nn.ReLU(),
            nn.Flatten(),
            nn.Linear(feature_dim * feature_dim * 2 * width, linear_size),
            nn.BatchNorm1d(linear_size),
            nn.ReLU(),
            nn.Linear(linear_size, n_classes)
        )

    def forward(self, x):
        """Return network predictions for the supplied input batch."""
        return self.layers(x)

    class GowalConvSmall(nn.Module):
    """Construct the small convolutional classifier proposed by Gowal et al.

    The dataset option selects the affine input width for MNIST or CIFAR-10.
    Input shape and preprocessing must match that choice.

    Args:
        in_shape (tuple): Shape of the input tensor.
        n_classes (int): Number of output classes.
        dataset (str): The dataset being used. Can be 'mnist' or 'cifar10'.

    Attributes:
        layers (nn.Sequential): A sequential container of the layers in the network.

    Methods:
        forward(x):
            Defines the forward pass of the network.
    """
    def __init__(self, in_shape=(1, 28, 28), n_classes=10, dataset='mnist'):
        super(GowalConvSmall, self).__init__()
        assert in_shape[1] == in_shape[2], "We only support square inputs for now!"
        in_channels = in_shape[0]
        in_dim  = in_shape[1]
        if dataset == 'mnist':
            linear_in = 3200
        elif dataset == 'cifar10':
            linear_in = 4608
        self.layers = nn.Sequential(
            nn.Conv2d(in_channels, 16, kernel_size=(4,4), stride=2),
            nn.ReLU(inplace=True),
            nn.Conv2d(16, 32, kernel_size=(4, 4), stride=1),
            nn.ReLU(inplace=True),
            nn.Flatten(),
            nn.Linear(linear_in, 100),
            nn.ReLU(inplace=True),
            nn.Linear(100, n_classes)
        )

    def forward(self, x):
        """Return network predictions for the supplied input batch."""
        return self.layers(x)

    class CNN_GLS(nn.Sequential):
    """Construct the CNN3 or CNN5 architectures used for Gaussian loss smoothing.

    These models omit BatchNorm and embedded input normalisation. Depth and
    width select the variants reported in Table 9 of Balauca et al. (2025):
    https://arxiv.org/abs/2403.07095.

    ``depth=3, width=8`` gives CNN3; width=2 gives CNN3Tiny.
    ``depth=5, width=16`` gives CNN5; width=64 gives CNN5L.
    Inputs must already be normalised by the data loader.

    Args:
        in_shape (tuple[int, ...]): Input shape (channels, height, width), excluding the batch dimension.
        n_classes (int): Number of output classes.
        depth (int): Architecture depth; supported values are 3 and 5.
        width (int): Number of units or base channels in the hidden layers.
        linear_size (int): Number of units in the hidden affine layer of the CNN5 variant.
        initialization (str): Weight initialisation: 'default' or 'kaiming_uniform'.
    """
    def __init__(self, in_shape=(1, 28, 28), n_classes=10, depth=5,
                 width=16, linear_size=512, initialization='default'):
        if depth not in (3, 5) or width < 1:
            raise ValueError('depth must be 3 or 5 and width must be positive')
        if initialization not in ('default', 'kaiming_uniform'):
            raise ValueError('unsupported GLS initialization')
        layers = [nn.Conv2d(in_shape[0], width, 5, stride=2, padding=2), nn.ReLU(),
                  nn.Conv2d(width, width, 4, stride=2, padding=1), nn.ReLU()]
        if depth == 5:
            layers += [nn.Conv2d(width, 2 * width, 4, stride=2, padding=1), nn.ReLU()]
        with torch.no_grad():
            flat_size = nn.Sequential(*layers)(torch.zeros(1, *in_shape)).numel()
        layers += [nn.Flatten()]
        if depth == 5:
            layers += [nn.Linear(flat_size, linear_size), nn.ReLU()]
            flat_size = linear_size
        layers += [nn.Linear(flat_size, n_classes)]
        super().__init__(*layers)
        if initialization == 'kaiming_uniform':
            for layer in self:
                if isinstance(layer, (nn.Conv2d, nn.Linear)):
                    nn.init.kaiming_uniform_(layer.weight)

    class CNN6_SetBased(nn.Module):
    """Construct the Cnn6 architecture used for set-based training.

    The model combines four convolutional layers with BatchNorm and ReLU,
    followed by three affine layers. Its layout follows the interpretation of
    Appendix A of Koller et al. (2025) used by CTRAIN.

    Args:
        in_shape (tuple[int, ...]): Input shape (channels, height, width), excluding the batch dimension.
        n_classes (int): Number of output classes.
    """
    def __init__(self, in_shape=(1, 28, 28), n_classes=10):
        super().__init__()
        channels, height, width = in_shape
        layers = []
        for output, kernel, stride in [(32, 3, 1), (32, 4, 2), (64, 3, 1), (64, 4, 2)]:
            layers.extend([nn.Conv2d(channels, output, kernel, stride, padding=1),
                           nn.BatchNorm2d(output), nn.ReLU()])
            height = (height + 2 - kernel) // stride + 1
            width = (width + 2 - kernel) // stride + 1
            channels = output
        if min(height, width) <= 0:
            raise ValueError('Input spatial size too small for Cnn6')
        layers.extend([nn.Flatten(), nn.Linear(channels * height * width, 512),
                       nn.BatchNorm1d(512), nn.ReLU(), nn.Linear(512, 512),
                       nn.BatchNorm1d(512), nn.ReLU(), nn.Linear(512, n_classes)])
        self.layers = nn.Sequential(*layers)

    def forward(self, x):
        """Return network predictions for the supplied input batch."""
        return self.layers(x)

    class MNISTSwapGraph(nn.Module):
    """Evaluate both operand orders through one shared MNIST addition classifier.

    The graph concatenates the original and swapped predictions into 78 logits.
    This exposes relational properties while sharing all classifier parameters.

    Args:
        model (torch.nn.Module): PyTorch model to train or evaluate.
    """
    def __init__(self, model):
        super().__init__()
        self.model = model

    def forward(self, inputs):
        """Return network predictions for the supplied input batch."""
        swapped = torch.cat((inputs[:, 1:2], inputs[:, 0:1]), dim=1)
        return torch.cat((self.model(inputs), self.model(swapped)), dim=1)

    class ControllerMLP(nn.Sequential):
    """Construct a two-input, two-output controller with two 32-unit ReLU layers.

    Optional coordinate-feature initialisation exposes signed state features and
    starts with zero actions. All layers remain trainable; this initialisation
    does not itself ensure invariance under disturbances.

    Args:
        coordinate_features (bool): Whether to initialise signed coordinate features with zero
            output actions.
    """
    def __init__(self, coordinate_features=False):
        super().__init__(nn.Linear(2, 32), nn.ReLU(), nn.Linear(32, 32),
                         nn.ReLU(), nn.Linear(32, 2))
        if coordinate_features:
            with torch.no_grad():
                for layer in (self[0], self[2], self[4]):
                    layer.weight.zero_()
                    layer.bias.zero_()
                for coordinate in range(2):
                    for j, (sign, offset) in enumerate(((1., 0.), (-1., 0.),
                                                       (1., -.3), (-1., -.3))):
                        row = coordinate * 4 + j
                        self[0].weight[row, coordinate] = sign
                        self[0].bias[row] = offset
                self[2].weight.copy_(torch.eye(32))

    class ControllerDynamics(nn.Module):
    """Compose a controller with fixed additive state-transition dynamics.

    Inputs contain [x0, x1, w0, w1]. Outputs contain the two controller actions
    followed by successor states x_next = x + u + w. This graph lets output
    specifications constrain actions and next states with shared controller weights.

    Args:
        controller (torch.nn.Module): Controller mapping the two state coordinates to two actions.
    """
    def __init__(self, controller):
        super().__init__()
        self.controller = controller

    def forward(self, inputs):
        """Return network predictions for the supplied input batch."""
        action = self.controller(inputs[:, :2])
        successor = inputs[:, :2] + action + inputs[:, 2:]
        return torch.cat((action, successor), dim=1)
