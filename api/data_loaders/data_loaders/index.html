    Data loader API

    Data and models documents the shared image-loader options, splits, transforms, normalization metadata and architecture choices. For imported verification problems, see property import semantics.

    Load image datasets together with their input limits and normalisation metadata.

    Bound propagation requires feasible input ranges and, for normalised images, the statistics needed to transform raw-pixel perturbation radii. The returned loaders expose mean, std, min, max and normalised for this purpose.

    def download_tinyimagenet(data_root='./data'):
    """Download and extract TinyImageNet when the expected dataset files are missing.

    After extraction, validation images are moved into class subdirectories by
    create_val_img_folder. Existing complete dataset directories are reused.

    Args:
        data_root (str, optional): Root directory where the dataset will be stored.
    """
    url = 'http://cs231n.stanford.edu/tiny-imagenet-200.zip'
    dataset_dir = os.path.join(data_root, 'tiny-imagenet-200')
    zip_path = os.path.join(data_root, 'tiny-imagenet-200.zip')

    # Check if the dataset directory exists and contains the expected subdirectories
    expected_subdirs = ['train', 'val', 'test', 'wnids.txt', 'words.txt']
    if not os.path.exists(dataset_dir) or not all(os.path.exists(os.path.join(dataset_dir, subdir)) for subdir in expected_subdirs):
        os.makedirs(data_root, exist_ok=True)
        print(f'Downloading TinyImageNet dataset from {url}...')
        response = requests.get(url, stream=True)
        with open(zip_path, 'wb') as f:
            for chunk in response.iter_content(chunk_size=1024):
                if chunk:
                    f.write(chunk)
        print('Extracting TinyImageNet dataset...')
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            zip_ref.extractall(data_root)
        os.remove(zip_path)
        print('TinyImageNet dataset downloaded and extracted successfully.')
        create_val_img_folder(dataset_dir)
        print('Validation images separated into subfolders.')
    else:
        print('TinyImageNet dataset already exists and is complete.')

    def load_cifar10(batch_size=64, normalise=True, train_transforms=(transforms.RandomHorizontalFlip(), transforms.RandomCrop(32, 2, padding_mode='edge')),
                  val_split=True, data_root='./data', *, workers=None, download=True,
                  selection_split=None, validation_fraction=.2, seed=None, smoke=False,
                  input_size=None, normalisation_stats=None):
    """Load CIFAR-10 and attach the metadata required for bound propagation.

    Training transformations apply only to training images. A training holdout
    uses evaluation transformations, so validation does not augment its inputs.

    Args:
        batch_size (int): Number of images per loader batch.
        normalise (bool): Whether to apply channelwise dataset normalisation.
        train_transforms (Sequence): Image transformations preceding tensor conversion on
            training data.
        val_split (bool): Whether to create a training holdout when selection_split is None.
        data_root (str): Dataset storage directory.
        workers (int, optional): Number of loader workers; None uses the dataset default.
        download (bool): Whether to download missing data.
        selection_split (str, optional): 'validation' uses a training holdout, 'test' reuses
            the test loader, and 'none' disables selection.
            None follows val_split.
        validation_fraction (float): Training fraction reserved for a holdout, in (0, 1).
        seed (int, optional): Seed for selecting holdout indices.
        smoke (bool): Whether to return two synthetic images per split for execution checks.
        input_size (int, optional): Positive centre-crop size; None retains the dataset
            resolution.
        normalisation_stats (dict, optional): Custom channel mean and std with
            normalise=True.

    Returns:
        (tuple): (train_loader, test_loader) when selection_split is None and
            val_split=False; otherwise (train_loader, selection_loader, test_loader).
            The selection loader is None when selection is disabled. All loaders
            expose mean, std, min, max and normalised attributes.
    """

                            val_split, data_root, workers, download, selection_split,
                            validation_fraction, seed, smoke, input_size, normalisation_stats)

    def load_gtsrb(batch_size=64, normalise=True, train_transforms=[transforms.RandomRotation(10), transforms.RandomResizedCrop(size=(30, 30), scale=(0.85, 1.0)), transforms.RandomAffine(degrees=0, translate=(0.1, 0.1), shear=15),], val_split=True, data_root='./data'):
    """Load GTSRB with image transforms, track-based validation and normalisation metadata.

    Args:
        batch_size (int, optional): The number of samples per batch to load.
        normalise (bool, optional): Whether to normalise the dataset using predefined mean
            and std values.
        train_transforms (list, optional): List of torchvision transforms to apply to the
            training data. Default includes random rotation,
            resized crop, and affine transformations.
        val_split (bool, optional): Whether to split the training data into training and
            validation sets.
        data_root (str, optional): Root directory where the dataset is stored or will be
            downloaded.

    Returns:
        (tuple): If val_split is True, returns a tuple of DataLoader objects (train_loader, val_loader, test_loader). Otherwise, returns (train_loader, test_loader).

    The DataLoader objects returned have additional attributes:
        - mean (torch.Tensor): The mean used for normalisation.
        - std (torch.Tensor): The standard deviation used for normalisation.
        - min (torch.Tensor): The minimum value in the dataset.
        - max (torch.Tensor): The maximum value in the dataset.
        - normalised (bool): Whether the dataset was normalised.
    """
    if normalise:
        mean = torch.tensor([0.3403, 0.3121, 0.3214])
        std =  torch.tensor([0.2724, 0.2608, 0.2669])
    else:
        mean = torch.tensor([0, 0, 0])
        std = torch.tensor([1, 1, 1])
    if normalise:
        train_transform = transforms.Compose([
            *train_transforms,
            transforms.Resize((30, 30)),
            transforms.ToTensor(),
            transforms.Normalize(mean, std),
        ])
        test_transform = transforms.Compose([
            transforms.ToTensor(),
            transforms.Resize((30, 30)),
            transforms.Normalize(mean, std),
        ])
    else:
        train_transform = transforms.Compose([
            *train_transforms,
            transforms.Resize((30, 30)),
            transforms.ToTensor(),
        ])
        test_transform = transforms.Compose([
            transforms.Resize((30, 30)),
            transforms.ToTensor(),
        ])

    train_dataset_ori = datasets.GTSRB(root=data_root, split='train', download=True, transform=train_transform)
    test_dataset = datasets.GTSRB(root=data_root, split='test', download=True, transform=test_transform)

    if val_split:
        file_names = train_dataset_ori._samples
        track_id_to_sample = [name[0].split("/")[-2] + "_" + name[0].split("/")[-1].split("_")[0] for name in file_names]
        track_ids = set(track_id_to_sample)
        train_track_ids, val_track_ids = train_test_split(np.array(sorted(track_ids)), test_size=0.2, train_size=0.8)
        train_ids = [i for i in range(len(track_id_to_sample)) if track_id_to_sample[i] in train_track_ids]
        val_ids = [i for i in range(len(track_id_to_sample)) if track_id_to_sample[i] in val_track_ids]


        train_dataset = Subset(train_dataset_ori, train_ids)
        evaluation_dataset = copy.copy(train_dataset_ori)
        evaluation_dataset.transform = test_transform
        val_dataset = Subset(evaluation_dataset, val_ids)

        val_loader = DataLoader(val_dataset, batch_size=batch_size, shuffle=False)
        val_loader.mean, val_loader.std = mean, std

    else:
        train_dataset = train_dataset_ori


    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    test_loader = DataLoader(test_dataset, batch_size=batch_size, shuffle=False)

    # Initialize variables to track the min and max values
    min_val = ((torch.zeros(mean.shape) - mean) / std).view(1, *mean.shape, 1, 1)
    max_val = ((torch.ones(mean.shape) - mean) / std).view(1, *mean.shape, 1, 1)

    max_target = float('-inf')
    # Loop over the dataset
    for data, target in train_loader:
        max_target = torch.tensor(max(max_target, target.max().item()))


    print(f'GTSRB dataset - Min value: {min_val}, Max value: {max_val}, no_classes: {max_target}')

    train_loader.mean, train_loader.std, train_loader.min, train_loader.max, train_loader.normalised = mean, std, min_val, max_val, normalise
    test_loader.mean, test_loader.std, test_loader.min, test_loader.max, test_loader.normalised = mean, std, min_val, max_val, normalise

    if val_split:
        val_loader.mean, val_loader.std, val_loader.min, val_loader.max, val_loader.normalised = mean, std, min_val, max_val, normalise
        return train_loader, val_loader, test_loader
    else:
        return train_loader, test_loader

    def load_vnnlib_problem(model_path, specification_path, *, name=None,
                        input_shape=None, output_semantics='unsafe', expected_sha256=None):
    """Load an ONNX network and VNNLIB specification for property-aware training.

    The loader constructs the desired output formula and finite input boxes,
    converts the model and checks predictions at box endpoints, midpoints and
    128 deterministic random inputs per box. File paths and hashes record the
    provenance of the resulting training problem.

    Args:
        model_path (str | pathlib.Path): Local single-input/output ONNX model.
        specification_path (str | pathlib.Path): Local bounded-box VNNLIB file.
        name (str | None): Property name, defaulting to the specification stem.
        input_shape (tuple[int, ...] | None): Shape without batch. Inferred from
            static ONNX dimensions, otherwise defaults to a vector. Override
            this when dynamic image dimensions need an explicit shape.
        output_semantics (str): `'unsafe'` complements source output assertions;
            `'safe'` treats them as desired conditions. Defaults to `'unsafe'`.
        expected_sha256 (dict | None): Optional expected digests with exactly
            the keys `'model'` and `'specification'`. Checked before file parsing.

    Returns:
        (tuple): `(model, property_spec, provenance)`. Model is trainable on CPU
            in eval mode. Provenance records resolved paths, SHA256 hashes, input
            shape and output semantics. Recorded hashes alone do not authenticate
            an upstream source. Input normalisation remains the caller's choice.

    Raises:
        ValueError: Hashes, shapes, input dtype or VNNLIB constructs unsupported.
        AssertionError: Converted predictions fail ONNX Runtime parity.

    ONNX float32/float64 inputs and batched vector outputs are supported. Bound
    operator support is checked later by `PropertyModelWrapper`. See
    `load_vnnlib_property` for the supported specification subset.
    """
    import onnx
    paths = dict(model=Path(model_path), specification=Path(specification_path))
    hashes = {key: hashlib.sha256(path.read_bytes()).hexdigest() for key, path in paths.items()}
    if expected_sha256 is not None and expected_sha256 != hashes:
        raise ValueError('Expected model/specification SHA256 hashes do not match')
    graph = onnx.load(paths['model'])
    initializers = {value.name for value in graph.graph.initializer}
    inputs = [value for value in graph.graph.input if value.name not in initializers]
    if len(inputs) != 1:
        raise ValueError('Property problems require a single ONNX input')
    dtype = {onnx.TensorProto.FLOAT: torch.float32,
             onnx.TensorProto.DOUBLE: torch.float64}.get(inputs[0].type.tensor_type.elem_type)
    if dtype is None:
        raise ValueError('ONNX input must be float32 or float64')
    dimensions = [dim.dim_value for dim in inputs[0].type.tensor_type.shape.dim][1:]
    if input_shape is None and dimensions and all(dim > 0 for dim in dimensions):
        input_shape = tuple(dimensions)
    spec, width = _read_vnnlib(paths['specification'], name, input_shape, output_semantics)
    generator = torch.Generator().manual_seed(17)
    batches = []
    for box in spec.domains:
        random = box.lower + torch.rand((128, *box.lower.shape), generator=generator,
                                       dtype=torch.double) * (box.upper - box.lower)
        batches.append(torch.cat((torch.stack((box.lower, box.upper, (box.lower + box.upper) / 2)), random)))
    probes = torch.cat(batches).to(dtype)
    model = load_onnx_model(paths['model'], probes)
    with torch.no_grad():
        if model(probes).shape != (len(probes), width):
            raise ValueError('ONNX output width must match the declared VNNLIB outputs')
    provenance = dict(paths={key: str(path.resolve()) for key, path in paths.items()},
                      sha256=hashes, input_shape=list(box.lower.shape), output_semantics=output_semantics)
    return model, spec, provenance

    def acasxu_property(case, domain):
    """Construct the safe ACAS Xu property 7 or 8 over the supplied input box.

    Args:
        case (str): Either ``'7'`` or ``'8'``. Lower output scores select actions.
        domain (Box): Released input coordinates, without extra normalisation.

    Returns:
        (PropertySpec): Property 7 excludes globally minimal strong-left/right
            scores. Property 8 requires one allowed advisory (clear of conflict or
            weak-left) to beat all forbidden ones. Both exclude unsafe ties.
    """
    if not isinstance(domain, Box) or domain.lower.shape != (5,):
        raise ValueError('ACAS Xu properties require a five-coordinate input box')
    y = Outputs()
    if case == '7':
        formula = And(*(Or(*(y[i] > y[j] for j in range(5) if i != j)) for i in (3, 4)))
    elif case == '8':
        formula = Or(*(And(*(y[i] < y[j] for j in (2, 3, 4))) for i in (0, 1)))
    else:
        raise ValueError("Supported ACAS Xu cases are '7' and '8'")
    return PropertySpec(f'acasxu_{case}', formula, [domain])

    """Load the source-checked ACAS Xu model and safe output specification for a repair
    example.

    train and certify using `PropertyModelWrapper` afterwards.
    For caller-selected networks or properties, use `load_vnnlib_problem`
    instead; this function supplies only the two original experiment presets.
    """

        raise ValueError("Supported ACAS Xu cases are '7' and '8'")

    manifest = json.loads((directory / 'manifest.json').read_text())
    if manifest.get('source') != SOURCE or not re.fullmatch(r'[0-9a-f]{40}', manifest.get('revision', '')):
        raise ValueError('Artifact manifest requires the expected source and pinned commit SHA')

    if set(hashes) != set(FILES):
        raise ValueError('Artifact manifest must contain all four required files')
    for filename in FILES:
        if hashlib.sha256((directory / filename).read_bytes()).hexdigest() != hashes[filename]:
            raise ValueError(f'Artifact hash mismatch: {filename}')
    box, unsafe = _read_source_spec(directory / f'prop_{case}.vnnlib')
    _crosscheck(case, unsafe)

                                          dtype=torch.double) * (box.upper - box.lower)

    with torch.no_grad():
        if model(probes).shape != (len(probes), 5):
            raise ValueError('ACAS Xu model must return five advisory scores per input')
